// 1. Find students older than 22
db.students.find({
  age: { $gt: 22 }
})


// 2. Find students enrolled in React
db.students.find({
  Courses: "Reactjs"
})


// 3. Find students whose name starts with "S"
db.students.find({
  name: { $regex: /^S/i }
})


// 4. Find students whose age is either 18 or 21
db.students.find({
  age: { $in: [18, 21] }
})


// 5. Find students who are NOT using Gmail
db.students.find({
  email: { $not: /@gmail\.com$/ }
})


// 6. Find students enrolled in React AND older than 20
db.students.find({
  Courses: "Reactjs",
  age: { $gt: 20 }
})


// 7. Find students enrolled in React OR Node.js
db.students.find({
  Courses: {
    $in: ["Reactjs", "Nodejs"]
  }
})


// 8. BONUS: Names containing "x"
db.students.find({
  name: { $regex: /x/i }
})


// 8. BONUS: Emails ending in ".edu"
db.students.find({
  email: { $regex: /\.edu$/i }
})
